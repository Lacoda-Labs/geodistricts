

export PROJECT_ID='geodistricts'
export STATIC_MAPS_BUCKET='geodistricts_map_data'
export CDN_HOSTNAME='maps-cdn.geodistricts.org'

gcloud storage buckets create gs://geodistricts_map_data \
  --project=geodistricts \
  --location=US \
  --default-storage-class=STANDARD \
  --uniform-bucket-level-access

  gcloud storage buckets add-iam-policy-binding gs://geodistricts_map_data \
  --member=allUsers \
  --role=roles/storage.objectViewer

  export STATIC_MAPS_CDN_BASE='https://YOUR_CDN_HOST/public-maps'
export STATIC_MAPS_GCS_PREFIX='gs://YOUR_BUCKET/public-maps'
npm run build:static-maps-cdn && npm run build:static-maps-cdn -- --upload

# ssl provision status
gcloud compute ssl-certificates describe geodistricts-static-maps-ssl-cert \
  --global --format='yaml(managed.status,managed.domainStatus)'
